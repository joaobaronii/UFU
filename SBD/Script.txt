CREATE TABLE Premio (
    codigo_premio SMALLINT,
    tipo VARCHAR(50) NOT NULL,
    data_entrega DATE,
    CONSTRAINT pk_premio PRIMARY KEY (codigo_premio)
);


CREATE TABLE Desconto (
    codigo_desconto SMALLINT,
    valor NUMERIC(10,2) NOT NULL,
    CONSTRAINT pk_desconto PRIMARY KEY (codigo_desconto)
);


CREATE TABLE Pacote (
    codigo_pacote SMALLINT,
    nome VARCHAR(100) NOT NULL,
    preco NUMERIC(10,2) NOT NULL,
    funcoes VARCHAR(255),
    CONSTRAINT pk_pacote PRIMARY KEY (codigo_pacote)
);


CREATE TABLE Upgrade (
    codigo_upgrade SMALLINT,
    valor NUMERIC(10,2) NOT NULL,
    beneficios VARCHAR(255),
    CONSTRAINT pk_upgrade PRIMARY KEY (codigo_upgrade)
);


CREATE TABLE Conta_Usuario (
    codigo_conta SMALLINT,
    cpf VARCHAR(11) UNIQUE NOT NULL,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    telefone VARCHAR(15),
    endereco VARCHAR(255),
    data_nascimento DATE,
    nome_usuario VARCHAR(50) NOT NULL,
    senha VARCHAR(50) NOT NULL,
    nome_conta VARCHAR(100),
    codigo_premio SMALLINT,
    CONSTRAINT pk_conta_usuario PRIMARY KEY (codigo_conta),
    CONSTRAINT fk_conta_usuario_premio FOREIGN KEY (codigo_premio) REFERENCES Premio(codigo_premio)
);


CREATE TABLE Empresarial (
    codigo_conta SMALLINT,
    cnpj VARCHAR(14)  UNIQUE NOT NULL,
    nome VARCHAR(100) NOT NULL,
    codigo_pacote SMALLINT,
    codigo_desconto SMALLINT,
    CONSTRAINT pk_empresarial PRIMARY KEY (codigo_conta),
    CONSTRAINT fk_empresarial_conta FOREIGN KEY (codigo_conta) REFERENCES Conta_Usuario(codigo_conta),
    CONSTRAINT fk_empresarial_pacote FOREIGN KEY (codigo_pacote) REFERENCES Pacote(codigo_pacote),
    CONSTRAINT fk_empresarial_desconto FOREIGN KEY (codigo_desconto) REFERENCES Desconto(codigo_desconto)
);


CREATE TABLE Pessoal (
    codigo_conta SMALLINT,
    nro_de_amigos SMALLINT DEFAULT 0,
    CONSTRAINT pk_pessoal PRIMARY KEY (codigo_conta),
    CONSTRAINT fk_pessoal_conta FOREIGN KEY (codigo_conta) REFERENCES Conta_Usuario(codigo_conta)
);


CREATE TABLE Grupo_de_Amigos (
    codigo_grupo SMALLINT,
    nome VARCHAR(100) NOT NULL,
    descricao VARCHAR(255),
    codigo_conta SMALLINT,
    CONSTRAINT pk_grupo_amigos PRIMARY KEY (codigo_grupo),
    CONSTRAINT fk_grupo_pessoal FOREIGN KEY (codigo_conta) REFERENCES Pessoal(codigo_conta)
);


CREATE TABLE Participa (
    codigo_conta SMALLINT,
    codigo_grupo SMALLINT,
    CONSTRAINT pk_participa PRIMARY KEY (codigo_conta, codigo_grupo),
    CONSTRAINT fk_participa_conta FOREIGN KEY (codigo_conta) REFERENCES Pessoal(codigo_conta),
    CONSTRAINT fk_participa_grupo FOREIGN KEY (codigo_grupo) REFERENCES Grupo_de_Amigos(codigo_grupo)
);


CREATE TABLE Postagem (
    codigo_postagem SMALLINT,
    tipo VARCHAR(50) NOT NULL,
    numero_curtidas SMALLINT DEFAULT 0,
    data_postagem DATE,
    codigo_conta SMALLINT,
    CONSTRAINT pk_postagem PRIMARY KEY (codigo_postagem),
    CONSTRAINT fk_postagem_conta FOREIGN KEY (codigo_conta) REFERENCES Conta_Usuario(codigo_conta)
);


CREATE TABLE Pagina (
    codigo_conta SMALLINT,
    nome VARCHAR(100) NOT NULL,
    nro_de_visitas SMALLINT DEFAULT 0,
    codigo_upgrade SMALLINT,
    CONSTRAINT pk_pagina PRIMARY KEY (codigo_conta),
    CONSTRAINT uq_pagina_conta_nome UNIQUE (codigo_conta, nome),
    CONSTRAINT fk_pagina_conta FOREIGN KEY (codigo_conta) REFERENCES Conta_Usuario(codigo_conta),
    CONSTRAINT fk_pagina_upgrade FOREIGN KEY (codigo_upgrade) REFERENCES Upgrade(codigo_upgrade)
);



CREATE TABLE Pagina_Propaganda (
    codigo_conta SMALLINT,
    nome VARCHAR(100),
    taxa_rejeicao NUMERIC(5,2),
    taxa_conversao NUMERIC(5,2),
    taxa_clique NUMERIC(5,2),
    CONSTRAINT pk_pagina_propaganda PRIMARY KEY (codigo_conta, nome),
    CONSTRAINT fk_pagina_propaganda FOREIGN KEY (codigo_conta, nome) REFERENCES Pagina(codigo_conta, nome)
);


CREATE TABLE Visivel (
    codigo_grupo SMALLINT,
    codigo_conta SMALLINT,
    nome VARCHAR(100),
    CONSTRAINT pk_visivel PRIMARY KEY (codigo_grupo, codigo_conta, nome),
    CONSTRAINT fk_visivel_grupo FOREIGN KEY (codigo_grupo) REFERENCES Grupo_de_Amigos(codigo_grupo),
    CONSTRAINT fk_visivel_pagina FOREIGN KEY (codigo_conta, nome) REFERENCES Pagina(codigo_conta, nome)
);


CREATE TABLE Amizade (
    codigo_amizade SMALLINT,
    codigo_amigo SMALLINT,
    status VARCHAR(20),
    CONSTRAINT pk_amizade PRIMARY KEY (codigo_amizade, codigo_amigo),
    CONSTRAINT fk_amizade_conta1 FOREIGN KEY (codigo_amizade) REFERENCES Conta_Usuario(codigo_conta),
    CONSTRAINT fk_amizade_conta2 FOREIGN KEY (codigo_amigo) REFERENCES Conta_Usuario(codigo_conta)
);


CREATE TABLE Acao (
    codigo_acao SMALLINT,
    descricao VARCHAR(255) NOT NULL,
    CONSTRAINT pk_acao PRIMARY KEY (codigo_acao)
);
